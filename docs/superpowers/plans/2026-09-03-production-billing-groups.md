**Goal:** Allow CrewHead and COO to explicitly group existing event records for future common invoicing, including an optional cross-project/Job Number exception, without changing the events, their hours, or existing invoices.

## Contract decisions

1. A group has a UUID and a name. An event belongs to at most one group. Membership points to the original event UUID, not its numeric client projection or Job Number.
2. Local-preview keys are explicitly `local:<event.id>`. Never send them to Supabase or fall back to a numeric ID when a remote UUID is missing.
3. A small singleton revision serializes this infrequently used feature. A stale edit is rejected, even if another group changed. Prefer this conservative rule over per-group locking complexity for v1.
4. The mutation replaces one target group's membership in one transaction; moving selected events out of other groups needs explicit confirmation. Detaching never deletes events.
5. Every affected event is rechecked against the version seen by the editor. Project/name/date edits during review force a refresh and confirmation.
6. Read access to a group must not expose hidden member IDs, other people's hours, totals, or private event details. The membership SELECT policy inherits visibility from `events`.
7. Empty groups may be deleted. A linked event must be detached before deleting that event; add a clear error, not an implicit cascade. The future invoice-intake migration must add an `ON DELETE RESTRICT` reference before it starts referencing groups.
8. A stable request UUID is reused after an ambiguous network response. A changed payload gets a new UUID only after a definitive outcome or a refreshed, explicitly reviewed draft.
9. Grouping does not imply the eventual invoice has one valid recipient. The current invoice resolver uses the project's client; the later invoice-intake design must verify the actual firm being invoiced, retain allocation by event/Job Number, and reject double billing.

```ts
import type { Event, Role } from '../../types';

export type BillingGroup = { id: string; name: string; eventIds: string[] };
export type BillingSnapshot = { revision: number | null; groups: BillingGroup[] };
export type BillingEventVersion = { id: string; projectKey: string; version: string };
export type BillingScope = {
  source: 'local' | 'supabase'; userId: string | null;
  profileId: string | null; role: Role;
};
export type SaveBillingGroup = {
  requestId: string; groupId: string; name: string;
  eventIds: string[]; expectedRevision: number;
  eventVersions: Record<string, string>;
  confirmCrossProject: boolean; confirmMoves: boolean;
  deleteGroup: boolean;
};
export type BillingMutationResult = {
  requestId: string; groupId: string; revision: number;
};
export class BillingError extends Error {
  constructor(public kind: 'conflict' | 'denied' | 'invalid' | 'ambiguous', message: string) { super(message); }
}

export const canManageBillingGroups = (role: Role) => role === 'crewhead' || role === 'coo';
export const billingEventKey = (event: Event, source: BillingScope['source']): string => {
  if (source === 'local') return `local:${event.id}`;
  if (!event.supabaseId) throw new Error('Akci chybí serverová identita. Obnovte data.');
  return event.supabaseId;
};
export const billingEventVersion = (event: Event, source: BillingScope['source']): BillingEventVersion => ({
  id: billingEventKey(event, source),
  projectKey: `${event.projectId ?? ''}|${event.job ?? ''}`,
  version: source === 'supabase'
    ? event.updatedAt ?? ''
    : JSON.stringify([event.name, event.projectId ?? null, event.job, event.startDate, event.endDate]),
});
export function validateSelection(
  role: Role, groupId: string, ids: string[], groups: BillingGroup[],
  events: BillingEventVersion[], cross: boolean, moves: boolean,
): void {
  if (!canManageBillingGroups(role)) throw new Error('Společnou fakturaci může měnit pouze produkce.');
  if (new Set(ids).size !== ids.length) throw new Error('Akce je ve výběru vícekrát.');
  const eventById = new Map(events.map(event => [event.id, event]));
  if (ids.some(id => !eventById.get(id)?.version)) throw new Error('Výběr akcí není aktuální. Obnovte data.');
  if (new Set(ids.map(id => eventById.get(id)!.projectKey)).size > 1 && !cross) {
    throw new Error('Potvrďte společnou fakturaci přes více projektů.');
  }
  if (groups.some(group => group.id !== groupId && group.eventIds.some(id => ids.includes(id))) && !moves) {
    throw new Error('Potvrďte přesun z jiné fakturační skupiny.');
  }
}
export function replaceMembership(groups: BillingGroup[], target: BillingGroup): BillingGroup[] {
  const selected = new Set(target.eventIds);
  return [
    ...groups.filter(group => group.id !== target.id).map(group => ({
      ...group, eventIds: group.eventIds.filter(id => !selected.has(id)),
    })),
    { ...target, eventIds: [...target.eventIds] },
  ];
}
export function buildGroupCommand(
  scope: BillingScope, snapshot: BillingSnapshot, group: BillingGroup,
  events: Event[], confirmations: { cross: boolean; moves: boolean },
  requestId: string, deleteGroup = false,
): SaveBillingGroup {
  const versions = events.filter(event => scope.source === 'local' || Boolean(event.supabaseId))
    .map(event => billingEventVersion(event, scope.source));
  validateSelection(scope.role, group.id, group.eventIds, snapshot.groups, versions,
    confirmations.cross, confirmations.moves);
  if (snapshot.revision === null) throw new Error('Chybí verze fakturačních skupin.');
  const name = group.name.trim();
  if (!deleteGroup && (name.length === 0 || name.length > 120)) throw new Error('Název musí mít 1 až 120 znaků.');
  const previous = snapshot.groups.find(item => item.id === group.id)?.eventIds ?? [];
  const affected = new Set([...previous, ...group.eventIds]);
  const eventVersions = Object.fromEntries(versions.filter(event => affected.has(event.id))
    .map(event => [event.id, event.version]));
  if (Object.keys(eventVersions).length !== affected.size || Object.values(eventVersions).some(version => !version)) {
    throw new Error('Výběr akcí není aktuální. Obnovte data.');
  }
  return { requestId, groupId: group.id, name, eventIds: [...group.eventIds].sort(),
    expectedRevision: snapshot.revision, eventVersions,
    confirmCrossProject: confirmations.cross, confirmMoves: confirmations.moves, deleteGroup };
}
```

```sql
begin;
create table public.billing_groups (
  id uuid primary key,
  name text not null check (char_length(btrim(name)) between 1 and 120)
);
create table public.billing_group_members (
  event_id uuid primary key references public.events(id) on delete restrict,
  group_id uuid not null references public.billing_groups(id) on delete restrict
);
create index billing_group_members_group_idx on public.billing_group_members(group_id);
create table public.billing_group_state (
  singleton boolean primary key default true check (singleton),
  revision integer not null default 0 check (revision >= 0)
);
insert into public.billing_group_state(singleton) values (true);
create table public.billing_group_requests (
  request_id uuid primary key,
  actor_id uuid not null references auth.users(id) on delete cascade,
  payload jsonb not null,
  result jsonb not null,
  created_at timestamptz not null default now()
);
create index billing_group_requests_actor_idx on public.billing_group_requests(actor_id);

create function public.can_manage_billing_groups()
returns boolean language sql stable security invoker set search_path = '' as $$
  select (select auth.uid()) is not null and (
    public.has_role((select auth.uid()), 'crewhead'::public.app_role)
    or public.has_role((select auth.uid()), 'coo'::public.app_role)
  );
$$;
revoke all on function public.can_manage_billing_groups() from public, anon;
grant execute on function public.can_manage_billing_groups() to authenticated;

alter table public.billing_groups enable row level security;
alter table public.billing_group_members enable row level security;
alter table public.billing_group_state enable row level security;
alter table public.billing_group_requests enable row level security;
revoke all on public.billing_groups, public.billing_group_members,
  public.billing_group_state, public.billing_group_requests from anon, authenticated;
grant select, insert, update, delete on public.billing_groups, public.billing_group_members to authenticated;
grant select, update on public.billing_group_state to authenticated;
grant select, insert on public.billing_group_requests to authenticated;

create policy billing_member_read on public.billing_group_members for select to authenticated
using (exists (select 1 from public.events e where e.id = event_id));
create policy billing_member_write on public.billing_group_members for all to authenticated
using ((select public.can_manage_billing_groups()))
with check ((select public.can_manage_billing_groups())
  and exists (select 1 from public.events e where e.id = event_id));
create policy billing_group_read on public.billing_groups for select to authenticated
using ((select public.can_manage_billing_groups()) or exists (
  select 1 from public.billing_group_members m where m.group_id = billing_groups.id
));
create policy billing_group_write on public.billing_groups for all to authenticated
using ((select public.can_manage_billing_groups()))
with check ((select public.can_manage_billing_groups()));
create policy billing_state_manage on public.billing_group_state for all to authenticated
using ((select public.can_manage_billing_groups()))
with check ((select public.can_manage_billing_groups()));
create policy billing_request_own on public.billing_group_requests for all to authenticated
using (actor_id = (select auth.uid()) and (select public.can_manage_billing_groups()))
with check (actor_id = (select auth.uid()) and (select public.can_manage_billing_groups()));

create function public.guard_billing_group_write()
returns trigger language plpgsql security invoker set search_path = '' as $$
begin
  if not public.can_manage_billing_groups()
    or coalesce(current_setting('app.billing_group_write', true), '') <> 'atomic' then
    raise exception 'billing_group_atomic_required' using errcode = '42501';
  end if;
  if tg_op = 'DELETE' then return old; end if;
  return new;
end;
$$;
revoke all on function public.guard_billing_group_write() from public, anon, authenticated;
create trigger billing_groups_guard before insert or update or delete on public.billing_groups
for each row execute function public.guard_billing_group_write();
create trigger billing_members_guard before insert or update or delete on public.billing_group_members
for each row execute function public.guard_billing_group_write();
create trigger billing_state_guard before update on public.billing_group_state
for each row execute function public.guard_billing_group_write();
create trigger billing_requests_guard before insert on public.billing_group_requests
for each row execute function public.guard_billing_group_write();

create function public.read_billing_groups()
returns jsonb language sql stable security invoker set search_path = '' as $$
  select jsonb_build_object(
    'revision', (select revision from public.billing_group_state where singleton),
    'groups', coalesce((select jsonb_agg(jsonb_build_object(
      'id', g.id, 'name', g.name,
      'event_ids', coalesce((select jsonb_agg(m.event_id order by m.event_id)
        from public.billing_group_members m where m.group_id = g.id), '[]'::jsonb)
    ) order by g.id) from public.billing_groups g), '[]'::jsonb)
  );
$$;
revoke all on function public.read_billing_groups() from public, anon;
grant execute on function public.read_billing_groups() to authenticated;

create function public.save_billing_group_atomic(
  p_request_id uuid, p_group_id uuid, p_name text, p_event_ids uuid[],
  p_expected_revision integer, p_event_versions jsonb,
  p_confirm_cross_project boolean, p_confirm_moves boolean, p_delete boolean
)
returns jsonb language plpgsql security invoker set search_path = '' as $$
declare
  v_revision integer;
  v_payload jsonb;
  v_previous public.billing_group_requests%rowtype;
  v_result jsonb;
  v_affected uuid[];
  v_seen integer := 0;
  v_event record;
  v_old_marker text := current_setting('app.billing_group_write', true);
begin
  if not public.can_manage_billing_groups() then
    raise exception 'billing_group_unauthorized' using errcode = '42501';
  end if;
  if p_request_id is null or p_group_id is null or p_event_ids is null
    or p_expected_revision is null or p_delete is null
    or jsonb_typeof(p_event_versions) is distinct from 'object'
    or exists (select 1 from unnest(p_event_ids) x where x is null)
    or cardinality(p_event_ids) <> (select count(distinct x) from unnest(p_event_ids) x) then
    raise exception 'billing_group_invalid_input' using errcode = '22023';
  end if;
  if not p_delete and (p_name is null or char_length(btrim(p_name)) not between 1 and 120) then
    raise exception 'billing_group_invalid_input' using errcode = '22023';
  end if;
  v_payload := jsonb_build_object('group', p_group_id, 'name', p_name,
    'events', p_event_ids, 'revision', p_expected_revision, 'versions', p_event_versions,
    'cross', p_confirm_cross_project, 'moves', p_confirm_moves, 'delete', p_delete);
  select revision into strict v_revision from public.billing_group_state
    where singleton for update;
  select * into v_previous from public.billing_group_requests where request_id = p_request_id;
  if found then
    if v_previous.actor_id <> auth.uid() or v_previous.payload <> v_payload then
      raise exception 'billing_group_request_mismatch' using errcode = '22023';
    end if;
    return v_previous.result;
  end if;
  if v_revision <> p_expected_revision then
    raise exception 'billing_group_conflict' using errcode = '40001';
  end if;
  select coalesce(array_agg(distinct x order by x), '{}'::uuid[]) into v_affected from (
    select unnest(p_event_ids) x
    union select event_id from public.billing_group_members where group_id = p_group_id
  ) affected;
  if (select count(*) from jsonb_object_keys(p_event_versions)) <> cardinality(v_affected) then
    raise exception 'billing_group_conflict' using errcode = '40001';
  end if;
  for v_event in select e.id, e.updated_at from public.events e
    where e.id = any(v_affected) order by e.id for share loop
    v_seen := v_seen + 1;
    if v_event.updated_at is distinct from (p_event_versions ->> v_event.id::text)::timestamptz then
      raise exception 'billing_group_conflict' using errcode = '40001';
    end if;
  end loop;
  if v_seen <> cardinality(v_affected) then
    raise exception 'billing_group_event_unavailable' using errcode = '42501';
  end if;
  if (select count(distinct (e.project_id, e.job_number)) from public.events e
    where e.id = any(p_event_ids)) > 1 and not coalesce(p_confirm_cross_project, false) then
    raise exception 'billing_group_cross_project_confirmation' using errcode = '22023';
  end if;
  if exists (select 1 from public.billing_group_members
    where event_id = any(p_event_ids) and group_id <> p_group_id)
    and not coalesce(p_confirm_moves, false) then
    raise exception 'billing_group_move_confirmation' using errcode = '22023';
  end if;
  if p_delete and (cardinality(p_event_ids) <> 0 or exists (
    select 1 from public.billing_group_members where group_id = p_group_id
  )) then
    raise exception 'billing_group_not_empty' using errcode = '23503';
  end if;
  perform set_config('app.billing_group_write', 'atomic', true);
  if p_delete then
    delete from public.billing_groups where id = p_group_id;
    if not found then raise exception 'billing_group_missing' using errcode = '22023'; end if;
  else
    insert into public.billing_groups(id, name) values (p_group_id, btrim(p_name))
      on conflict(id) do update set name = excluded.name;
    delete from public.billing_group_members
      where group_id = p_group_id and not (event_id = any(p_event_ids));
    insert into public.billing_group_members(event_id, group_id)
      select x, p_group_id from unnest(p_event_ids) x
      on conflict(event_id) do update set group_id = excluded.group_id;
  end if;
  update public.billing_group_state set revision = revision + 1 where singleton
    returning revision into v_revision;
  v_result := jsonb_build_object('request_id', p_request_id, 'group_id', p_group_id, 'revision', v_revision);
  insert into public.billing_group_requests(request_id, actor_id, payload, result)
    values(p_request_id, auth.uid(), v_payload, v_result);
  perform set_config('app.billing_group_write', coalesce(v_old_marker, ''), true);
  return v_result;
exception when others then
  perform set_config('app.billing_group_write', coalesce(v_old_marker, ''), true);
  raise;
end;
$$;
revoke all on function public.save_billing_group_atomic(uuid,uuid,text,uuid[],integer,jsonb,boolean,boolean,boolean)
  from public, anon;
grant execute on function public.save_billing_group_atomic(uuid,uuid,text,uuid[],integer,jsonb,boolean,boolean,boolean)
  to authenticated;
commit;
```

```ts
import { getLocalAppState } from '../../lib/app-data';
import { BillingError } from './billing-groups.model';
import { billingEventVersion, canManageBillingGroups, replaceMembership, validateSelection,
 type BillingScope, type BillingSnapshot, type SaveBillingGroup, type BillingMutationResult } from './billing-groups.model';
let state: BillingSnapshot = { revision: 0, groups: [] };
const requests = new Map<string,{ actor: string; payload: string; result: BillingMutationResult }>();
export function readLocalBillingGroups(scope: BillingScope): BillingSnapshot {
 if (canManageBillingGroups(scope.role)) return structuredClone(state);
 const data = getLocalAppState();
 const visible = new Set(data.events.filter(event =>
   event.status === 'upcoming' || event.status === 'full' || Boolean(scope.profileId && (
     data.eventCrewAssignments.some(item => item.eventId === event.id && item.contractorProfileId === scope.profileId)
     || data.eventApplications.some(item => item.eventId === event.id && item.contractorProfileId === scope.profileId)
     || data.timelogs.some(item => item.eid === event.id && item.contractorProfileId === scope.profileId)
   ))).map(event => `local:${event.id}`));
 return { revision: null, groups: state.groups.map(group => ({
   ...group, eventIds: group.eventIds.filter(id => visible.has(id)),
 })).filter(group => group.eventIds.length > 0) };
}
export function saveLocalBillingGroup(scope: BillingScope, command: SaveBillingGroup): BillingMutationResult {
 if (!canManageBillingGroups(scope.role)) throw new BillingError('denied','Společnou fakturaci může měnit pouze produkce.');
 const actor = scope.userId ?? scope.profileId ?? 'local-production';
 const payload = JSON.stringify(command);
 const previous = requests.get(command.requestId);
 if (previous) {
   if (previous.actor !== actor || previous.payload !== payload) throw new BillingError('invalid','Požadavek má jiné údaje.');
   return { ...previous.result };
 }
 if (command.expectedRevision !== state.revision) throw new BillingError('conflict','Data se mezitím změnila. Obnovte výběr.');
 const versions = getLocalAppState().events.map(event => billingEventVersion(event,'local'));
 const affected = new Set([...(state.groups.find(group => group.id === command.groupId)?.eventIds ?? []),...command.eventIds]);
 if (Object.keys(command.eventVersions).length !== affected.size || [...affected].some(id =>
   !command.eventVersions[id] || command.eventVersions[id] !== versions.find(event => event.id === id)?.version)) {
   throw new BillingError('conflict','Akce se mezitím změnila. Obnovte výběr.');
 }
 validateSelection(scope.role,command.groupId,command.eventIds,state.groups,versions,command.confirmCrossProject,command.confirmMoves);
 const name = command.name.trim();
 if (!command.deleteGroup && (!name || name.length > 120)) throw new BillingError('invalid','Název musí mít 1 až 120 znaků.');
 if (command.deleteGroup && (command.eventIds.length || state.groups.find(group => group.id === command.groupId)?.eventIds.length)) {
   throw new BillingError('invalid','Smazat lze pouze prázdnou skupinu.');
 }
 if (command.deleteGroup && !state.groups.some(group => group.id === command.groupId)) throw new BillingError('invalid','Skupina už neexistuje.');
 state = { revision: command.expectedRevision + 1, groups: command.deleteGroup
   ? state.groups.filter(group => group.id !== command.groupId)
   : replaceMembership(state.groups,{ id: command.groupId,name,eventIds:command.eventIds }) };
 const result = { requestId: command.requestId,groupId: command.groupId,revision: state.revision! };
 requests.set(command.requestId,{ actor,payload,result });
 return { ...result };
}
```
